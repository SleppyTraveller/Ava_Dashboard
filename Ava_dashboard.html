    const imageList = ['ava1.jpg', 'ava2.jpg', 'ava3.jpg', 'ava4.jpg'];

    let currentImage = 0;

    setInterval(() => {

      currentImage = (currentImage + 1) % imageList.length;

      document.getElementById('rotatingImage').src = imageList[currentImage];

    }, 20000);

    async function loadCSV() {

      const response = await fetch('data.csv');

      const csvText = await response.text();

      const lines = csvText.trim().split('\n');

      const tbody = document.getElementById('dataBody');

      const categoryFilter = document.getElementById('categoryFilter');

      const searchInput = document.getElementById('searchInput');

      const data = lines.map(line => line.split(',').map(cell => cell.replace(/"/g, '')));

      const categories = new Set();

      function renderTable(filter = '', search = '') {

        tbody.innerHTML = '';

        data.forEach(row => {

          const [title, link, category, date, notes] = row;

          if ((filter === '' || category === filter) && title.toLowerCase().includes(search.toLowerCase())) {

            const tr = document.createElement('tr');

            tr.innerHTML = `

              <td>${title}</td>

              <td>${category}</td>

              <td><a href="${link}" target="_blank">Åbn</a></td>

              <td>${date}</td>

              <td>${notes}</td>

            `;

            tbody.appendChild(tr);

          }

        });

      }

      data.forEach(row => categories.add(row[2]));

      categories.forEach(cat => {

        const opt = document.createElement('option');

        opt.value = cat;

        opt.textContent = cat;

        categoryFilter.appendChild(opt);

      });

      categoryFilter.addEventListener('change', () => renderTable(categoryFilter.value, searchInput.value));

      searchInput.addEventListener('input', () => renderTable(categoryFilter.value, searchInput.value));

      renderTable();

    }

    loadCSV();

  </script></body>
